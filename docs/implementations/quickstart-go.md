# Go quickstart

Sign and verify an N-AALP object with the Go reference implementation.

```go
package main

import (
	"fmt"

	"github.com/bubblefish-tech/naalp_protocol/impl/go/cbor"
	"github.com/bubblefish-tech/naalp_protocol/impl/go/channels"
	"github.com/bubblefish-tech/naalp_protocol/impl/go/cose"
	"github.com/bubblefish-tech/naalp_protocol/impl/go/envelope"
	"github.com/cloudflare/circl/sign/mldsa/mldsa65"
)

func main() {
	// 1. A signing key (use a real 32-byte seed in production).
	var seed [mldsa65.SeedSize]byte
	pk, sk := mldsa65.NewKeyFromSeed(&seed)

	// 2. Build a Control/Hello object (channel 0x0000, kind 0, read_only, Public profile).
	obj := &envelope.Object{
		Kind:    0, // Hello
		Channel: 0, // Control
		Tier:    0,
		Signer:  pk.Bytes(),
		Created: 1785000000000,
		Effect:  0, // read_only
		Profile: uint64(cose.ProfilePublic),
		Body:    cbor.Tstr("hello"),
	}

	// 3. Sign: content-id-binds and signs the object.
	signed, err := envelope.Sign(obj, cose.MLDSA65Signer{SK: sk})
	if err != nil {
		panic(err)
	}

	// 4. Verify offline against the channel-surface kind validator.
	got, err := envelope.Verify(cose.ProfilePublic, cose.MLDSA65Verifier{PK: pk},
		channels.KindValidator, nil, signed)
	if err != nil {
		panic(err) // any tamper or wrong effect is a named error, fail-closed
	}
	fmt.Printf("verified: channel=%d kind=%d effect=%d bytes=%d\n",
		got.Channel, got.Kind, got.Effect, len(signed))
}
```

Run the tests and the byte-parity check:

```console
$ cd impl/go && GOWORK=off go test -race -count=1 ./...
$ bash scripts/verify.sh          # from the repo root: Go == Rust byte parity
```

A complete, every-byte-grounded object is generated by
`GOWORK=off go run ./cmd/naalp-worked-example` — see the [worked example](../examples/worked-object.md).
